### FASE 2 — Training Model 1 (MobileNetV2)

**Step 6 — Preprocessing khusus MobileNetV2:**
```python
import tensorflow as tf
from tensorflow.keras.applications.mobilenet_v2 import preprocess_input as mobilenet_preprocess
from tensorflow.keras.preprocessing.image import ImageDataGenerator

datagen_mobile = ImageDataGenerator(
    preprocessing_function=mobilenet_preprocess,
    rotation_range=20,
    width_shift_range=0.2,
    height_shift_range=0.2,
    zoom_range=0.2,
    horizontal_flip=True,
    brightness_range=[0.8, 1.2],
    validation_split=0.2
)

train_mobile = datagen_mobile.flow_from_directory(
    '/content/data_tilapia',
    target_size=(224, 224),
    batch_size=32,
    class_mode='binary',
    subset='training',
    seed=42
)

val_mobile = datagen_mobile.flow_from_directory(
    '/content/data_tilapia',
    target_size=(224, 224),
    batch_size=32,
    class_mode='binary',
    subset='validation',
    seed=42
)

print("Kelas:", train_mobile.class_indices)
print(f"Train: {train_mobile.samples} | Val: {val_mobile.samples}")
```

**Step 7 — Build & train MobileNetV2:**
```python
from tensorflow.keras.applications import MobileNetV2
from tensorflow.keras.models import Model
from tensorflow.keras.layers import Dense, GlobalAveragePooling2D, Dropout, BatchNormalization
from tensorflow.keras.optimizers import Adam
from tensorflow.keras.callbacks import EarlyStopping, ReduceLROnPlateau, ModelCheckpoint

base_mobile = MobileNetV2(weights='imagenet', include_top=False, input_shape=(224, 224, 3))
base_mobile.trainable = False

x = base_mobile.output
x = GlobalAveragePooling2D()(x)
x = BatchNormalization()(x)
x = Dropout(0.3)(x)
x = Dense(128, activation='relu')(x)
x = Dropout(0.2)(x)
output = Dense(1, activation='sigmoid')(x)

model_mobile = Model(inputs=base_mobile.input, outputs=output)
model_mobile.compile(
    optimizer=Adam(learning_rate=0.001),
    loss='binary_crossentropy',
    metrics=['accuracy']
)

callbacks_mobile = [
    EarlyStopping(monitor='val_accuracy', patience=5, restore_best_weights=True, verbose=1),
    ReduceLROnPlateau(monitor='val_loss', factor=0.5, patience=3, verbose=1),
    ModelCheckpoint(
        '/content/drive/MyDrive/tilapia_project/model_mobilenet_tilapia.keras',
        monitor='val_accuracy', save_best_only=True, verbose=1
    )
]

print("🚀 Training MobileNetV2...")
history_mobile = model_mobile.fit(
    train_mobile, epochs=30,
    validation_data=val_mobile,
    callbacks=callbacks_mobile
)

best_mobile = max(history_mobile.history['val_accuracy'])
print(f"✅ MobileNetV2 selesai! Best val_accuracy: {best_mobile*100:.2f}%")
```

**Step 9 — Build & train EfficientNetB0 (Phase 1):**
```python
from tensorflow.keras.applications import EfficientNetB0

base_eff = EfficientNetB0(weights='imagenet', include_top=False, input_shape=(224, 224, 3))
base_eff.trainable = False

x = base_eff.output
x = GlobalAveragePooling2D()(x)
x = BatchNormalization()(x)
x = Dropout(0.3)(x)
x = Dense(128, activation='relu')(x)

model_eff = Model(inputs=base_eff.input, outputs=output)
model_eff.compile(
    optimizer=Adam(learning_rate=0.001),
    loss='binary_crossentropy',
    metrics=['accuracy']
)

callbacks_eff = [
    EarlyStopping(monitor='val_accuracy', patience=5, restore_best_weights=True, verbose=1),
    ReduceLROnPlateau(monitor='val_loss', factor=0.5, patience=3, verbose=1),
    ModelCheckpoint(
        '/content/drive/MyDrive/tilapia_project/model_efficientnet_tilapia.keras',
        monitor='val_accuracy', save_best_only=True, verbose=1
    )
]

print("🚀 Training EfficientNetB0 Phase 1...")
history_eff = model_eff.fit(
    train_eff, epochs=30,
    validation_data=val_eff,
    callbacks=callbacks_eff
)

best_eff = max(history_eff.history['val_accuracy'])
print(f"✅ EfficientNetB0 selesai! Best val_accuracy: {best_eff*100:.2f}%")
```

**Step 10 — Fine-tuning EfficientNetB0 (Phase 2):**
```python
# Buka 30 layer terakhir
for layer in model_eff.layers[:-30]:
    layer.trainable = False
for layer in model_eff.layers[-30:]:
    layer.trainable = True

model_eff.compile(
    optimizer=Adam(learning_rate=0.0001),
    loss='binary_crossentropy',
    metrics=['accuracy']
)

callbacks_eff2 = [
    EarlyStopping(monitor='val_accuracy', patience=5, restore_best_weights=True, verbose=1),
    ModelCheckpoint(
        '/content/drive/MyDrive/tilapia_project/model_efficientnet_tilapia_final.keras',
        monitor='val_accuracy', save_best_only=True, verbose=1
    )
]

print("🚀 Fine-tuning EfficientNetB0 Phase 2...")
history_eff2 = model_eff.fit(
    train_eff, epochs=15,
    validation_data=val_eff,
    callbacks=callbacks_eff2
)

best_eff2 = max(history_eff2.history['val_accuracy'])
print(f"✅ Fine-tuning selesai! Best val_accuracy: {best_eff2*100:.2f}%")
```

### FASE 4 — Evaluasi & Perbandingan

**Step 11 — Evaluasi kedua model:**
```python
from sklearn.metrics import classification_report, confusion_matrix
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

# Evaluasi langsung
loss_m, acc_m = model_mobile.evaluate(val_mobile, verbose=0)
loss_e, acc_e = model_eff.evaluate(val_eff, verbose=0)

print("=" * 55)
print(f"{'Model':<25} {'Accuracy':>10} {'Loss':>10}")
print("=" * 55)
print(f"{'MobileNetV2':<25} {acc_m*100:>9.2f}% {loss_m:>10.4f}")
print(f"{'EfficientNetB0':<25} {acc_e*100:>9.2f}% {loss_e:>10.4f}")
print("=" * 55)

# Tentukan model terbaik
if acc_e >= acc_m:
    best_model = model_eff
    best_name = "EfficientNetB0"
    best_gen = val_eff
else:
    best_model = model_mobile
    best_name = "MobileNetV2"
    best_gen = val_mobile

print(f"\n🏆 Model terbaik: {best_name}")
```

**Step 12 — Confusion matrix model terbaik:**
```python
best_gen.reset()
y_pred = (best_model.predict(best_gen) > 0.5).astype(int).flatten()
y_true = best_gen.classes[:len(y_pred)]

print(f"\n📊 Classification Report — {best_name}:")
print(classification_report(y_true, y_pred, target_names=['Fresh', 'NonFresh']))

cm = confusion_matrix(y_true, y_pred)
plt.figure(figsize=(8, 6))
sns.heatmap(cm, annot=True, fmt='d', cmap='Blues',
            xticklabels=['Fresh', 'NonFresh'],
            yticklabels=['Fresh', 'NonFresh'])
plt.title(f'Confusion Matrix — {best_name}', fontsize=14, fontweight='bold')
plt.ylabel('Label Asli')
plt.xlabel('Prediksi Model')
plt.tight_layout()
plt.savefig('/content/drive/MyDrive/tilapia_project/confusion_matrix_best.png', dpi=150)
plt.show()
print("✅ Confusion matrix tersimpan ke Drive!")
```
